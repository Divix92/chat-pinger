cmake_minimum_required(VERSION 3.28)
project(chat-pinger-obs VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(WIN32)
  # OBS upstream bootstrap verifies every downloaded SDK archive with SHA-256.
  set(CMAKE_TLS_VERIFY ON)
  set(OS_WINDOWS TRUE)
  list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake/common" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/windows")
  include(buildspec)
endif()
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network WebSockets)
add_library(chat-pinger MODULE src/plugin.cpp)
target_link_libraries(chat-pinger PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network Qt6::WebSockets)
target_compile_definitions(chat-pinger PRIVATE QT_NO_KEYWORDS)
set_target_properties(chat-pinger PROPERTIES PREFIX "")
if(WIN32)
  # Standard OBS bundles Qt Core/Network/Widgets, but not WebSockets.
  install(FILES $<TARGET_FILE:Qt6::WebSockets> DESTINATION bin/64bit)
  install(TARGETS chat-pinger RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
  install(DIRECTORY data/ DESTINATION data/obs-plugins/chat-pinger)
else()
  install(TARGETS chat-pinger LIBRARY DESTINATION lib/obs-plugins)
  install(DIRECTORY data/ DESTINATION share/obs/obs-plugins/chat-pinger)
endif()
